# FASE 2 — Assembly (gcc -S)
# Cole aqui 5 a 10 linhas de soma.s que você considera mais
# interessantes e explique o que cada uma faz.

Trecho de soma.s:

main:
	pushq	%rbp
	movq	%rsp, %rbp
	movl	$10, %esi
	leaq	.LC0(%rip), %rax
	movq	%rax, %rdi
	call	printf@PLT
	movl	$0, %eax
	popq	%rbp
	ret

Explicação:
- pushq %rbp: salva a base da pilha.
- movq %rsp, %rbp: cria um novo quadro de pilha.
- movl $10, %esi: coloca o valor 10 em um registrador.
- leaq .LC0(%rip), %rax: carrega o endereço da string.
- movq %rax, %rdi: prepara argumento para printf.
- call printf@PLT: chama a função printf.
- movl $0, %eax: define retorno 0.
- popq %rbp: restaura a pilha.
- ret: encerra a função.
